\section{Descripción General}
\begin{itemize}[leftmargin=*]
    \item \textbf{Problema:} quien busca cuarto en Puno compara a mano anuncios dispersos, sin saber si la información es real ni qué reglas de convivencia tiene cada cuarto.
    \item \textbf{Objetivo del sistema:} ordenar los cuartos con un puntaje explicado de 0 a 100 (CASA SCORE) que combina precio, cercanía al lugar que elige el estudiante, seguridad, servicios y tipo de baño, según las prioridades de cada estudiante.
    \item \textbf{Principio de diseño:} todas las variables se llevan a una escala común de 0 a 100 antes de combinarse, las condiciones de convivencia restan puntos y todo puntaje se muestra desglosado.
\end{itemize}

\section{Reglas del Motor CASA SCORE}

\subsection{Criterios ponderados (escala 0 a 100)}
{\small
\noindent\begin{tabularx}{\textwidth}{@{}P{2.4cm} L@{}}
\toprule
\rowcolor{AzulClaro}
\textbf{Criterio} & \textbf{Regla de normalización} \\
\midrule
Precio ($p$) & $N_p = 100 \cdot \frac{850 - p}{850 - 140}$; 100 si $p \le$ S/ 140 y 0 si $p \ge$ S/ 850. \\
Cercanía & Distancia ($d$, de 50 a 3,000 m) y tiempo a pie ($t$, de 2 a 40 min) al punto de referencia del estudiante, al 50\,\% cada uno: $N_c = 0.5 \cdot 100\frac{3000-d}{3000-50} + 0.5 \cdot 100\frac{40-t}{40-2}$. \\
Seguridad & Nivel $s$ de 1 a 5: 1 punto base más 1 por cada característica declarada con foto (portón con llave propia, rejas en ventanas, iluminación exterior y cámara de vigilancia). $N_s = \frac{s}{5} \cdot 100$. \\
Servicios 24 h & Agua 40, luz 30 e internet 30 puntos. \\
Baño & Privado: 100. Compartido: 0. \\
\bottomrule
\end{tabularx}
}

\subsection{Puntaje final}
\[
\text{CASA SCORE} = \min\Big(100,\ \max\big(0,\ \tfrac{\sum_i w_i N_i}{\sum_i w_i} + B - P\big)\Big)
\]

{\small
\noindent\begin{tabularx}{\textwidth}{@{}L C{1.6cm} L C{1.6cm}@{}}
\toprule
\rowcolor{AzulClaro}
\textbf{Bonificación ($B$)} & \textbf{Puntos} & \textbf{Penalización ($P$)} & \textbf{Puntos} \\
\midrule
Ventana con luz natural directa & $+8$ & Hora de llegada hasta las 20:00 & $-20$ \\
Entrada libre 24 h con llave propia & $+5$ & Hora de llegada de 21:00 a 22:00 & $-12$ \\
Habitación de 16 m$^2$ o más & $+4$ & Hora de llegada después de las 22:00 & $-6$ \\
& & Prohibición de visitas & $-12$ \\
& & Prohibición de cocinar & $-8$ \\
& & Condiciones no declaradas en el anuncio & $-25$ \\
\bottomrule
\end{tabularx}
}

\section{Restricciones, Supuestos y Exclusiones}
\begin{itemize}[leftmargin=*]
    \item \textbf{Restricciones:} MVP en la semana 16 del semestre 2026-II; servicios en capa gratuita; solo Android.
    \item \textbf{Supuestos:} hay propietarios dispuestos a publicar; los usuarios tienen un celular Android con datos móviles y GPS.
    \item \textbf{Exclusiones:} sin comisiones por el alquiler, sin contratos, sin garantía de la seguridad física del inmueble y sin reemplazo de la visita al cuarto.
\end{itemize}
